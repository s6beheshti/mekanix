   297→// ─── Step 3: Matching ───
   298→export function Matching({ customer }: { customer: DemoUser }) {
   299→  const { go, params } = useApp();
   300→  const [techs, setTechs] = useState<Technician[] | null>(null);
   301→  const [phase, setPhase] = useState<"searching" | "results">("searching");
   302→  const [selected, setSelected] = useState<string | null>(null);
   303→
   304→  useEffect(() => {
   305→    const lat = 37.7749, lng = -122.4194;
   306→    api.listTechnicians({ lat, lng }).then((t) => {
   307→      // simulate searching delay
   308→      setTimeout(() => {
   309→        setTechs(t);
   310→        setPhase("results");
   311→      }, 1900);
   312→    });
   313→  }, []);
   314→
   315→  const ranked = (techs ?? []).map((t) => {
   316→    const km = t.lat && t.lng ? haversine({ lat: 37.7749, lng: -122.4194 }, { lat: t.lat, lng: t.lng }) : 99;
   317→    const eta = Math.max(5, Math.round((km / 35) * 60) + t.responseMins);
   318→    const score = t.rating * 20 - km * 0.4 + (t.verified ? 5 : 0) + t.completedJobs * 0.02;
   319→    return { tech: t, km, eta, score };
   320→  }).sort((a, b) => b.score - a.score);
   321→
   322→  const onSelect = (techId: string) => {
   323→    setSelected(techId);
   324→    go("technician-profile", { requestId: params.requestId, technicianId: techId });
   325→  };
   326→
   327→  return (
   328→    <div className="space-y-5">
   329→      <StepHeader step={3} total={3} title="Matching Technicians" subtitle="We found qualified, verified specialists near you" onBack={() => go("describe")} />
   330→
   331→      <AnimatePresence mode="wait">
   332→        {phase === "searching" ? (
   333→          <motion.div key="searching" exit={{ opacity: 0 }} className="grid place-items-center py-16">
   334→            <div className="relative">
   335→              <motion.div
   336→                animate={{ scale: [1, 1.8, 1], opacity: [0.6, 0, 0.6] }}
   337→                transition={{ duration: 2, repeat: Infinity }}
   338→                className="absolute inset-0 rounded-full bg-amber"
   339→              />
   340→              <div className="relative grid size-20 place-items-center rounded-full border-2 border-amber/30 bg-amber/10">
   341→                <Search className="size-8 text-amber" />
   342→              </div>
   343→            </div>
   344→            <p className="mt-6 font-display text-lg font-semibold">Searching nearby technicians…</p>
   345→            <p className="text-sm text-muted-foreground">Analyzing distance, expertise, availability & ratings</p>
   346→            <div className="mt-4 flex gap-6 text-[11px] text-muted-foreground">
   347→              <span className="inline-flex items-center gap-1"><NavIcon className="size-3 text-amber" /> Distance</span>
   348→              <span className="inline-flex items-center gap-1"><Star className="size-3 text-amber" /> Rating</span>
   349→              <span className="inline-flex items-center gap-1"><BadgeCheck className="size-3 text-amber" /> Certifications</span>
   350→              <span className="inline-flex items-center gap-1"><Clock className="size-3 text-amber" /> ETA</span>
   351→            </div>
   352→          </motion.div>
   353→        ) : (
   354→          <motion.div key="results" initial={{ opacity: 0 }} animate={{ opacity: 1 }}>
   355→            {ranked.length === 0 ? (
   356→              <EmptyState icon={Search} title="No technicians available" description="Try expanding your urgency or location. Emergency dispatch available 24/7." action={<Button onClick={() => go("describe")} variant="outline">Back to details</Button>} />
   357→            ) : (
   358→              <div className="grid gap-3 lg:grid-cols-2">
   359→                {ranked.map((r, i) => (
   360→                  <TechnicianCard
   361→                    key={r.tech.id}
   362→                    tech={r.tech}
   363→                    distanceKm={r.km}
   364→                    etaMins={r.eta}
   365→                    rank={i + 1}
   366→                    selected={selected === r.tech.id}
   367→                    onSelect={() => onSelect(r.tech.id)}
   368→                  />
   369→                ))}
   370→              </div>
   371→            )}
   372→          </motion.div>
   373→        )}
   374→      </AnimatePresence>
   375→    </div>
   376→  );
   377→}

   379→// ─── Technician Profile ───
   380→export function TechnicianProfileView({ customer }: { customer: DemoUser }) {
   381→  const { go, params, back } = useApp();
   382→  const [tech, setTech] = useState<Technician | null>(null);
   383→  const [requesting, setRequesting] = useState(false);
   384→
   385→  useEffect(() => {
   386→    if (params.technicianId) api.getTechnician(params.technicianId).then(setTech);
   387→  }, [params.technicianId]);
   388→
   389→  const request = async () => {
   390→    setRequesting(true);
   391→    try {
   392→      // assign the technician to the request + create a job in REQUESTED state
   393→      const sr = await fetch(`/api/service-requests`).then((r) => r.json()).then((list: ServiceRequest[]) => list.find((x) => x.id === params.requestId));
   394→      if (!sr) throw new Error("Request not found");
   395→      // create job
   396→      const res = await fetch("/api/jobs", { method: "POST" as any }).catch(() => null);
   397→      // The job creation needs a dedicated endpoint; use the matching+status flow instead:
   398→      // We'll create the job by patching the request's matchedTech and creating via a dedicated endpoint.
   399→      const jobRes = await fetch(`/api/service-requests/${sr.id}/assign`, {
   400→        method: "POST",
   401→        headers: { "Content-Type": "application/json" },
   402→        body: JSON.stringify({ technicianId: params.technicianId }),
   403→      });
   404→      const job = await jobRes.json();
   405→      if (!job?.id) throw new Error("Failed to assign technician");
   406→      toast.success(`${tech?.user.name} has been notified`);
   407→      go("track", { jobId: job.id });
   408→    } catch (e: any) {
   409→      toast.error(e.message ?? "Could not assign technician");
   410→    } finally {
   411→      setRequesting(false);
   412→    }
   413→  };
   414→
   415→  if (!tech) return <div className="h-64 rounded-xl bg-muted/60 mk-shimmer" />;
   416→
   417→  return (
   418→    <div className="space-y-5">
   419→      <button onClick={back} className="inline-flex items-center gap-1 text-sm text-muted-foreground hover:text-foreground">
   420→        <ArrowLeft className="size-4" /> Back to results
   421→      </button>
   422→
   423→      {/* Header */}
   424→      <div className="relative overflow-hidden rounded-xl border border-border bg-card p-5">
   425→        <div className="absolute -right-16 -top-16 size-48 rounded-full bg-amber/10 blur-3xl" />
   426→        <div className="relative flex flex-col gap-4 sm:flex-row sm:items-center">
   427→          <div className="relative shrink-0">
   428→            <div className="grid size-20 place-items-center overflow-hidden rounded-2xl border border-border bg-muted">
