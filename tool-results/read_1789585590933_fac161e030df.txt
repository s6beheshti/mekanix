   295→// ─── Step 3: Matching ───
   296→export function Matching({ customer }: { customer: DemoUser }) {
   297→  const { go, params } = useApp();
   298→  const [techs, setTechs] = useState<Technician[] | null>(null);
   299→  const [phase, setPhase] = useState<"searching" | "results">("searching");
   300→  const [selected, setSelected] = useState<string | null>(null);
   301→
   302→  useEffect(() => {
   303→    const lat = 37.7749, lng = -122.4194;
   304→    api.listTechnicians({ lat, lng }).then((t) => {
   305→      // simulate searching delay
   306→      setTimeout(() => {
   307→        setTechs(t);
   308→        setPhase("results");
   309→      }, 1900);
   310→    });
   311→  }, []);
   312→
   313→  const ranked = (techs ?? []).map((t) => {
   314→    const km = t.lat && t.lng ? haversine({ lat: 37.7749, lng: -122.4194 }, { lat: t.lat, lng: t.lng }) : 99;
   315→    const eta = Math.max(5, Math.round((km / 35) * 60) + t.responseMins);
   316→    const score = t.rating * 20 - km * 0.4 + (t.verified ? 5 : 0) + t.completedJobs * 0.02;
   317→    return { tech: t, km, eta, score };
   318→  }).sort((a, b) => b.score - a.score);
   319→
   320→  const onSelect = (techId: string) => {
   321→    setSelected(techId);
   322→    go("technician-profile", { requestId: params.requestId, technicianId: techId });
   323→  };
   324→
   325→  return (
   326→    <div className="space-y-5">
   327→      <StepHeader step={3} total={3} title="Matching Technicians" subtitle="We found qualified, verified specialists near you" onBack={() => go("describe")} />
   328→
   329→      <AnimatePresence mode="wait">
   330→        {phase === "searching" ? (
   331→          <motion.div key="searching" exit={{ opacity: 0 }} className="grid place-items-center py-16">
   332→            <div className="relative">
   333→              <motion.div
   334→                animate={{ scale: [1, 1.8, 1], opacity: [0.6, 0, 0.6] }}
   335→                transition={{ duration: 2, repeat: Infinity }}
   336→                className="absolute inset-0 rounded-full bg-amber"
   337→              />
   338→              <div className="relative grid size-20 place-items-center rounded-full border-2 border-amber/30 bg-amber/10">
   339→                <Search className="size-8 text-amber" />
   340→              </div>
   341→            </div>
   342→            <p className="mt-6 font-display text-lg font-semibold">Searching nearby technicians…</p>
   343→            <p className="text-sm text-muted-foreground">Analyzing distance, expertise, availability & ratings</p>
   344→            <div className="mt-4 flex gap-6 text-[11px] text-muted-foreground">
   345→              <span className="inline-flex items-center gap-1"><NavIcon className="size-3 text-amber" /> Distance</span>
   346→              <span className="inline-flex items-center gap-1"><Star className="size-3 text-amber" /> Rating</span>
   347→              <span className="inline-flex items-center gap-1"><BadgeCheck className="size-3 text-amber" /> Certifications</span>
   348→              <span className="inline-flex items-center gap-1"><Clock className="size-3 text-amber" /> ETA</span>
   349→            </div>
   350→          </motion.div>
   351→        ) : (
   352→          <motion.div key="results" initial={{ opacity: 0 }} animate={{ opacity: 1 }}>
   353→            {ranked.length === 0 ? (
   354→              <EmptyState icon={Search} title="No technicians available" description="Try expanding your urgency or location. Emergency dispatch available 24/7." action={<Button onClick={() => go("describe")} variant="outline">Back to details</Button>} />
   355→            ) : (
   356→              <div className="grid gap-3 lg:grid-cols-2">
   357→                {ranked.map((r, i) => (
   358→                  <TechnicianCard
   359→                    key={r.tech.id}
   360→                    tech={r.tech}
   361→                    distanceKm={r.km}
   362→                    etaMins={r.eta}
   363→                    rank={i + 1}
   364→                    selected={selected === r.tech.id}
   365→                    onSelect={() => onSelect(r.tech.id)}
   366→                  />
   367→                ))}
   368→              </div>
   369→            )}
   370→          </motion.div>
   371→        )}
   372→      </AnimatePresence>
   373→    </div>
   374→  );
   375→}

   377→// ─── Technician Profile ───
   378→export function TechnicianProfileView({ customer }: { customer: DemoUser }) {
   379→  const { go, params, back } = useApp();
   380→  const [tech, setTech] = useState<Technician | null>(null);
   381→  const [requesting, setRequesting] = useState(false);
   382→
   383→  useEffect(() => {
   384→    if (params.technicianId) api.getTechnician(params.technicianId).then(setTech);
   385→  }, [params.technicianId]);
   386→
   387→  const request = async () => {
   388→    setRequesting(true);
   389→    try {
   390→      // assign the technician to the request + create a job in REQUESTED state
   391→      const sr = await fetch(`/api/service-requests`).then((r) => r.json()).then((list: ServiceRequest[]) => list.find((x) => x.id === params.requestId));
   392→      if (!sr) throw new Error("Request not found");
   393→      // create job
   394→      const res = await fetch("/api/jobs", { method: "POST" as any }).catch(() => null);
   395→      // The job creation needs a dedicated endpoint; use the matching+status flow instead:
   396→      // We'll create the job by patching the request's matchedTech and creating via a dedicated endpoint.
   397→      const jobRes = await fetch(`/api/service-requests/${sr.id}/assign`, {
   398→        method: "POST",
   399→        headers: { "Content-Type": "application/json" },
   400→        body: JSON.stringify({ technicianId: params.technicianId }),
   401→      });
   402→      const job = await jobRes.json();
   403→      if (!job?.id) throw new Error("Failed to assign technician");
   404→      toast.success(`${tech?.user.name} has been notified`);
   405→      go("track", { jobId: job.id });
   406→    } catch (e: any) {
   407→      toast.error(e.message ?? "Could not assign technician");
   408→    } finally {
   409→      setRequesting(false);
   410→    }
   411→  };
   412→
   413→  if (!tech) return <div className="h-64 rounded-xl bg-muted/60 mk-shimmer" />;
   414→
   415→  return (
   416→    <div className="space-y-5">
   417→      <button onClick={back} className="inline-flex items-center gap-1 text-sm text-muted-foreground hover:text-foreground">
   418→        <ArrowLeft className="size-4" /> Back to results
   419→      </button>
   420→
   421→      {/* Header */}
   422→      <div className="relative overflow-hidden rounded-xl border border-border bg-card p-5">
   423→        <div className="absolute -right-16 -top-16 size-48 rounded-full bg-amber/10 blur-3xl" />
   424→        <div className="relative flex flex-col gap-4 sm:flex-row sm:items-center">
   425→          <div className="relative shrink-0">
   426→            <div className="grid size-20 place-items-center overflow-hidden rounded-2xl border border-border bg-muted">
   427→              {tech.user.avatar ? <img src={tech.user.avatar} alt={tech.user.name} className="size-full object-cover" /> : <span className="font-display text-2xl">{tech.user.name[0]}</span>}
   428→            </div>
