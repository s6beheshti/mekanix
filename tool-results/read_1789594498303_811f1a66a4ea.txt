    29→    29→// ─── Step 1: Select Machine ───
    30→    30→export function RequestType({ customer }: { customer: DemoUser }) {
    31→    31→  const { go, params, machineMode } = useApp();
    32→    32→  const { t, isFa, type: typeLabel } = useT();
    33→    33→  const [vehicles, setVehicles] = useState<Vehicle[] | null>(null);
    34→    34→  const [selectedType, setSelectedType] = useState<string | null>(params.type ?? null);
    35→    35→
    36→    36→  const allowedTypes = typesForMode(machineMode);
    37→    37→
    38→    38→  useEffect(() => {
    39→    39→    if (customer.customer)
    40→    40→      api.listVehicles(customer.customer.id)
    41→    41→        .then((v) => setVehicles(v.filter((x) => allowedTypes.includes(x.type))))
    42→    42→        .catch(() => setVehicles([]));
    43→    43→  }, [customer, machineMode]);
    44→    44→
    45→    45→  const saved = vehicles ?? [];
    46→    46→  const hasSaved = saved.length > 0;
    47→    47→
    48→    48→  const proceed = (vehicle?: Vehicle, type?: string) => {
    49→    49→    go("describe", { vehicleId: vehicle?.id, type: type ?? vehicle?.type ?? selectedType ?? (machineMode === "heavy" ? "TRUCK" : "CAR") });
    50→    50→  };
    51→    51→
    52→    52→  const visibleMachineTypes = MACHINE_TYPES.filter((m) => allowedTypes.includes(m.slug));
    53→    53→
    54→    54→  return (
    55→    55→    <div className="space-y-5" dir={isFa ? "rtl" : "ltr"}>
    56→    56→      <StepHeader
    57→    57→        step={1}
    58→    58→        total={3}
    59→    59→        title={t("req.selectMachine")}
    60→    60→        subtitle={t("req.selectMachineSub")}
    61→    61→        onBack={() => go("home")}
    62→    62→      />
    63→    63→
    64→    64→      {hasSaved && (
    65→    65→        <section>
    66→    66→          <SectionHeader title={t("req.savedFleet")} subtitle={t("req.savedFleetSub")} />
    67→    67→          <div className="mt-3 grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
    68→    68→            {saved.map((v) => (
    69→    69→              <VehicleCard key={v.id} vehicle={v} selected={params.vehicleId === v.id} onSelect={() => proceed(v)} />
    70→    70→            ))}
    71→    71→          </div>
    72→    72→        </section>
    73→    73→      )}
    74→    74→
    75→    75→      <section>
    76→    76→        <SectionHeader
    77→    77→          title={hasSaved ? t("req.pickType") : t("req.whatNeeds")}
    78→    78→          subtitle={t("req.whatNeedsSub")}
    79→    79→        />
    80→    80→        <div className="mt-3 grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-5">
    81→    81→          {visibleMachineTypes.map((m) => (
    82→    82→            <motion.button
    83→    83→              key={m.slug}
    84→    84→              whileHover={{ y: -2 }}
    85→    85→              onClick={() => { setSelectedType(m.slug); proceed(undefined, m.slug); }}
    86→    86→              className={`group relative overflow-hidden rounded-xl border bg-card p-4 text-center mk-card-hover ${selectedType === m.slug ? "border-amber mk-amber-glow" : "border-border hover:border-amber/40"}`}
    87→    87→            >
    88→    88→              <div className={`mx-auto grid size-12 place-items-center rounded-xl border ${selectedType === m.slug ? "border-amber/40 bg-amber/10" : "border-border bg-background"}`}>
    89→    89→                <MekIcon name={m.icon} className={`size-6 ${selectedType === m.slug ? "text-amber" : "text-muted-foreground"}`} />
    90→    90→              </div>
    91→    91→              <p className="mt-2 font-display text-sm font-semibold">{typeLabel(m.slug)}</p>
    92→    92→            </motion.button>
    93→    93→          ))}
    94→    94→        </div>
    95→    95→      </section>
    96→    96→    </div>
    97→    97→  );
    98→    98→}
