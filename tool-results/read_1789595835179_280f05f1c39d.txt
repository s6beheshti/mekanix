    29→// ─── Step 1: Select Machine ───
    30→export function RequestType({ customer }: { customer: DemoUser }) {
    31→  const { go, params, machineMode } = useApp();
    32→  const { t, isFa, type: typeLabel } = useT();
    33→  const [vehicles, setVehicles] = useState<Vehicle[] | null>(null);
    34→  const [selectedType, setSelectedType] = useState<string | null>(params.type ?? null);
    35→
    36→  const allowedTypes = typesForMode(machineMode);
    37→
    38→  useEffect(() => {
    39→    if (customer.customer)
    40→      api.listVehicles(customer.customer.id)
    41→        .then((v) => setVehicles(v.filter((x) => allowedTypes.includes(x.type))))
    42→        .catch(() => setVehicles([]));
    43→  }, [customer, machineMode]);
    44→
    45→  const saved = vehicles ?? [];
    46→  const hasSaved = saved.length > 0;
    47→
    48→  const proceed = (vehicle?: Vehicle, type?: string) => {
    49→    go("describe", { vehicleId: vehicle?.id, type: type ?? vehicle?.type ?? selectedType ?? (machineMode === "heavy" ? "TRUCK" : "CAR") });
    50→  };
    51→
    52→  const visibleMachineTypes = MACHINE_TYPES.filter((m) => allowedTypes.includes(m.slug));
    53→
    54→  return (
    55→    <div className="space-y-5" dir={isFa ? "rtl" : "ltr"}>
    56→      <StepHeader
    57→        step={1}
    58→        total={3}
    59→        title={t("req.selectMachine")}
    60→        subtitle={t("req.selectMachineSub")}
    61→        onBack={() => go("home")}
    62→      />
    63→
    64→      {hasSaved && (
    65→        <section>
    66→          <SectionHeader title={t("req.savedFleet")} subtitle={t("req.savedFleetSub")} />
    67→          <div className="mt-3 grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
    68→            {saved.map((v) => (
    69→              <VehicleCard key={v.id} vehicle={v} selected={params.vehicleId === v.id} onSelect={() => proceed(v)} />
    70→            ))}
    71→          </div>
    72→        </section>
    73→      )}
    74→
    75→      <section>
    76→        <SectionHeader
    77→          title={hasSaved ? t("req.pickType") : t("req.whatNeeds")}
    78→          subtitle={t("req.whatNeedsSub")}
    79→        />
    80→        <div className="mt-3 grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-5">
    81→          {visibleMachineTypes.map((m) => (
    82→            <motion.button
    83→              key={m.slug}
    84→              whileHover={{ y: -2 }}
    85→              onClick={() => { setSelectedType(m.slug); proceed(undefined, m.slug); }}
    86→              className={`group relative overflow-hidden rounded-xl border bg-card p-4 text-center mk-card-hover ${selectedType === m.slug ? "border-amber mk-amber-glow" : "border-border hover:border-amber/40"}`}
    87→            >
    88→              <div className={`mx-auto grid size-12 place-items-center rounded-xl border ${selectedType === m.slug ? "border-amber/40 bg-amber/10" : "border-border bg-background"}`}>
    89→                <MekIcon name={m.icon} className={`size-6 ${selectedType === m.slug ? "text-amber" : "text-muted-foreground"}`} />
    90→              </div>
    91→              <p className="mt-2 font-display text-sm font-semibold">{typeLabel(m.slug)}</p>
    92→            </motion.button>
    93→          ))}
    94→        </div>
    95→      </section>
    96→    </div>
    97→  );
    98→}

   408→// ─── Step 3: Matching ───
   409→export function Matching({ customer }: { customer: DemoUser }) {
   410→  const { go, params } = useApp();
   411→  const { t, isFa, money } = useT();
   412→  const lang = isFa ? "fa" : "en";
   413→  const [techs, setTechs] = useState<Technician[] | null>(null);
   414→  const [phase, setPhase] = useState<"searching" | "results">("searching");
   415→  const [selected, setSelected] = useState<string | null>(null);
   416→
   417→  useEffect(() => {
   418→    const lat = 37.7749, lng = -122.4194;
   419→    api.listTechnicians({ lat, lng }).then((tList) => {
   420→      // simulate searching delay
   421→      setTimeout(() => {
   422→        setTechs(tList);
   423→        setPhase("results");
   424→      }, 1900);
   425→    });
   426→  }, []);
   427→
   428→  const ranked = (techs ?? []).map((tk) => {
   429→    const km = tk.lat && tk.lng ? haversine({ lat: 37.7749, lng: -122.4194 }, { lat: tk.lat, lng: tk.lng }) : 99;
   430→    const eta = Math.max(5, Math.round((km / 35) * 60) + tk.responseMins);
   431→    const score = tk.rating * 20 - km * 0.4 + (tk.verified ? 5 : 0) + tk.completedJobs * 0.02;
   432→    // Inspection fee ~ 30min at hourly rate; travel fee scales with distance
   433→    const inspectionFee = Math.round((tk.hourlyRate * 0.5) * 100) / 100;
   434→    const travelFee = Math.round((tk.travelFeeBase + km * 0.5) * 100) / 100;
   435→    return { tech: tk, km, eta, score, inspectionFee, travelFee };
   436→  }).sort((a, b) => b.score - a.score);
   437→
   438→  const onSelect = (techId: string) => {
   439→    setSelected(techId);
   440→    go("technician-profile", { requestId: params.requestId, technicianId: techId });
   441→  };
   442→
   443→  return (
   444→    <div className="space-y-5" dir={isFa ? "rtl" : "ltr"}>
   445→      <StepHeader
   446→        step={3}
   447→        total={3}
   448→        title={t("req.matching")}
   449→        subtitle={t("req.matchingSub")}
   450→        onBack={() => go("describe")}
   451→      />
   452→
   453→      <AnimatePresence mode="wait">
   454→        {phase === "searching" ? (
   455→          <motion.div key="searching" exit={{ opacity: 0 }} className="grid place-items-center py-16">
   456→            <div className="relative">
   457→              <motion.div
   458→                animate={{ scale: [1, 1.8, 1], opacity: [0.6, 0, 0.6] }}
   459→                transition={{ duration: 2, repeat: Infinity }}
   460→                className="absolute inset-0 rounded-full bg-amber"
   461→              />
   462→              <div className="relative grid size-20 place-items-center rounded-full border-2 border-amber/30 bg-amber/10">
